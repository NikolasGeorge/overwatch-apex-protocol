rule("Apex: Death - Respawn Apex from Fall")
{
	event
	{
		Player Died;
		Team 1;
		All;
	}

	conditions
	{
		Event Was Environment == True;
		Is Hero Being Played(Hero(Torbjörn), Team 1) == False;
		Number Of Living Players(Team 2) >= 4;
	}

	actions
	{
		Big Message(All Players(All Teams), Custom String("{1} The Apex teleported to spawn {0}", Hero Icon String(Hero Of(Event Player)),
			Ability Icon String(Hero Of(Event Player), Button(Ultimate))));
		Event Player.Apex_Heal_Ledge = True;
		Set Healing Received(Event Player, 6);
		Wait(0.025, Ignore Condition);
		Respawn(Event Player);
		Wait(0.025, Ignore Condition);
		Disallow Button(Event Player, Button(Ability 1));
		Disallow Button(Event Player, Button(Ability 2));
		Disallow Button(Event Player, Button(Ultimate));
		Set Invisible(Event Player, All);
		Set Player Health(Event Player, 1);
		Set Status(Event Player, Null, Rooted, 20);
		Wait(20, Ignore Condition);
		Event Player.Apex_Heal_Ledge = False;
		Set Invisible(Event Player, None);
		Set Player Health(Event Player, 500);
		Allow Button(Event Player, Button(Ability 1));
		Allow Button(Event Player, Button(Ability 2));
		Allow Button(Event Player, Button(Ultimate));
	}
}
