rule("Apex: Death - Respawn Apex from Fall")
{
	event
	{
		Player Died;
		Team 1;
		All;
	}

	conditions
	{
		Event Was Environment == True;
		Number Of Living Players(Team 2) > 3;
	}

	actions
	{
		If((Is Hero Being Played(Hero(Junkrat), Team 1) && Is Using Ultimate(Event Player)) == True);
			Abort;
		End;
		Big Message(All Players(All Teams), Custom String("{1} The Apex teleported to spawn {0}", Hero Icon String(Hero Of(Event Player)),
			Ability Icon String(Hero Of(Event Player), Button(Ultimate))));
		Set Objective Description(All Players(Team 2), Custom String("The Apex is Healing. Spend time on the Objective!"),
			Visible To and String);
		Event Player.Apex_Heal_Ledge = True;
		Set Healing Received(Event Player, 12);
		Wait(0.025, Ignore Condition);
		Respawn(Event Player);
		Wait(0.025, Ignore Condition);
		Call Subroutine(Disable_All_Buttons);
		Set Invisible(Event Player, All);
		Set Player Health(Event Player, 1);
		Set Status(Event Player, Null, Rooted, 30);
		Wait(30, Ignore Condition);
		Event Player.Apex_Heal_Ledge = False;
		Set Invisible(Event Player, None);
		Set Player Health(Event Player, Max Health(Event Player) * 0.750);
		Call Subroutine(Enable_All_Buttons);
		Set Objective Description(All Players(Team 2), Custom String("Hunt the Apex"), Visible To and String);
	}
}

rule("Apex: DPS")
{
	event
	{
		Player Dealt Damage;
		All;
		All;
	}

	conditions
	{
		Victim != Attacker;
	}

	actions
	{
		Destroy In-World Text(Event Player.DpsTextId);
		Event Player.DpsPostition = Eye Position(Victim) + Vector(0, 0.400, 0);
		If(Event Was Critical Hit == True);
			Create In-World Text(Event Player, Round To Integer(Event Damage, Up), Event Player.DpsPostition, 1.500 + Event Damage / 250,
				Clip Against Surfaces, Visible To Position and String, Color(Red), Default Visibility);
		Else If(Event Was Environment);
			Create In-World Text(Event Player, Custom String("GAME!"), Event Player.DpsPostition, 2 + Event Damage / 1000,
				Clip Against Surfaces, Visible To Position and String, Color(Orange), Default Visibility);
		Else;
			Create In-World Text(Event Player, Round To Integer(Event Damage, Up), Event Player.DpsPostition, 0.850 + Event Damage / 250,
				Clip Against Surfaces, Visible To Position and String, Color(White), Default Visibility);
		End;
		Event Player.DpsTextId = Last Text ID;
		Chase Player Variable At Rate(Event Player, DpsPostition, Eye Position(Victim) + Up, 0.900, None);
		Skip If(Event Player.DpsTime == 0, 1);
		Event Player.DpsDamage += Round To Integer(Event Damage, Up);
	}
}
